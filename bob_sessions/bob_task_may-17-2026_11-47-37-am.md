    def _scan_repository(self):
        """STEP 1: Scan entire repository structure"""
        if not self.repo_path.exists():
            print(f"⚠ Repository not found at {self.repo_path}")
            print("Creating sample structure for demonstration...")
            self._create_sample_structure()
            return
        
        # Count files and directories
        for root, dirs, files in os.walk(self.repo_path):
            self.stats.total_directories += len(dirs)
            self.stats.total_files += len(files)
            
            for file in files:
                if file.endswith(('.c', '.h', '.py', '.cpp', '.hpp')):
                    self.stats.source_files += 1
                elif file.endswith(('.md', '.txt', '.rst', '.sgml')):
                    self.stats.documentation_files += 1
    
    def _analyze_git_history(self):
        """STEP 2: Analyze git commits within analysis window"""
        if not (self.repo_path / ".git").exists():
            print("⚠ Not a git repository, using sample data...")
            self.stats.total_commits_in_window = 150
            self.stats.flagged_commits = 25
            return
        
        cutoff_date = self.config.get_cutoff_date()
        
        try:
            # Get commits within window
            if cutoff_date:
                since_arg = f"--since={cutoff_date.isoformat()}"
                result = subprocess.run(
                    ["git", "-C", str(self.repo_path), "log", since_arg, "--oneline"],
                    capture_output=True,
                    text=True,
                    check=True
                )
            else:
                result = subprocess.run(
                    ["git", "-C", str(self.repo_path), "log", "--oneline"],
                    capture_output=True,
                    text=True,
                    check=True
                )
            
            commits = result.stdout.strip().split('\n')
            self.stats.total_commits_in_window = len(commits)
            
            # Flag commits with reasoning keywords
            reasoning_keywords = [
                "because", "workaround", "temporary", "assume", "constraint",
                "limitation", "TODO", "FIXME", "HACK", "legacy", "backward compat"
            ]
            
            for commit in commits:
                if any(keyword in commit.lower() for keyword in reasoning_keywords):
                    self.stats.flagged_commits += 1
                    
        except subprocess.CalledProcessError:
            print("⚠ Git command failed, using sample data...")
            self.stats.total_commits_in_window = 150
            self.stats.flagged_commits = 25

class GraphBuilder:

    def __init__(self, uri: str = "bolt://localhost:7687", user: str = "neo4j", password: str = "password"):
        self.uri = uri
        self.user = user
        self.password = password
        self.driver = None
        
    def connect(self):
        """Connect to Neo4j"""
        if not NEO4J_AVAILABLE:
            raise RuntimeError("neo4j package not installed")
        
        try:
            self.driver = GraphDatabase.driver(self.uri, auth=(self.user, self.password))
            # Test connection
            with self.driver.session() as session:
                result = session.run("RETURN 1 as test")
                result.single()
            print(f"✓ Connected to Neo4j at {self.uri}")
            return True
        except Exception as e:
            print(f"✗ Failed to connect to Neo4j: {e}")
            print("\nTo start Neo4j with Docker:")
            print("  docker run -p 7474:7474 -p 7687:7687 -e NEO4J_AUTH=neo4j/password neo4j:latest")
            return False
    
    def close(self):
        """Close Neo4j connection"""
        if self.driver:
            self.driver.close()
    
    def clear_graph(self):
        """Delete all Nexus nodes and relationships"""
        with self.driver.session() as session:
            session.run("MATCH (n:DPR) DETACH DELETE n")
            session.run("MATCH (n:DecayAlert) DETACH DELETE n")
        print("✓ Cleared existing graph")

        with self.driver.session() as session:

                "CREATE INDEX dpr_component IF NOT EXISTS FOR (d:DPR) ON (d.component)",
                "CREATE INDEX dpr_blast_radius IF NOT EXISTS FOR (d:DPR) ON (d.blast_radius_estimate)",

                "CREATE INDEX dpr_within_window IF NOT EXISTS FOR (d:DPR) ON (d.within_window)",
                "CREATE INDEX decay_alert_decaying IF NOT EXISTS FOR (d:DecayAlert) ON (d.already_decaying)"
            ]
            for idx in indexes:
                session.run(idx)
        print("✓ Created indexes")
    
    def create_dpr_nodes(self, dprs: list):
        """Create DPR nodes in Neo4j"""
        with self.driver.session() as session:
            for dpr in dprs:
                session.run("""
                    CREATE (d:DPR {
                        dpr_id: $dpr_id,

        dpr_id = f"DPR-{dpr_match.group(1)}" if dpr_match else None
        
        # Extract component if present

function createGraph() {
    if (!graphData || !graphData.dprs) return;
    
    const dprs = graphData.dprs;

    // Calculate layout using force-directed algorithm (simplified)

        scene.add(label);

// Calculate node positions using force-directed layout
function calculateLayout(dprs, ctgEdges) {
    const positions = {};
    const radius = 40;
    
    // Group by component
    const components = {};
    dprs.forEach(dpr => {
        if (!components[dpr.component]) {
            components[dpr.component] = [];
        }
        components[dpr.component].push(dpr);
    });
    
    // Arrange components in a circle
    const componentNames = Object.keys(components);
    const angleStep = (Math.PI * 2) / componentNames.length;
    
    componentNames.forEach((component, compIndex) => {
        const compAngle = angleStep * compIndex;
        const compX = Math.cos(compAngle) * radius;
        const compZ = Math.sin(compAngle) * radius;
        
        // Arrange nodes within component
        const compNodes = components[component];
        const nodeRadius = 10;
        const nodeAngleStep = (Math.PI * 2) / compNodes.length;
        
        compNodes.forEach((dpr, nodeIndex) => {
            const nodeAngle = nodeAngleStep * nodeIndex;
            const x = compX + Math.cos(nodeAngle) * nodeRadius;
            const y = (Math.random() - 0.5) * 20;
            const z = compZ + Math.sin(nodeAngle) * nodeRadius;
            
            positions[dpr.dpr_id] = { x, y, z };
        });
    });
    
    return positions;
}

// Create a node mesh
function createNode(dpr, position) {
    const color = COLORS[dpr.blast_radius_estimate] || COLORS.low;
    
    // Size based on blast radius
    const sizeMap = { critical: 1.5, high: 1.2, medium: 1.0, low: 0.8 };
    const size = sizeMap[dpr.blast_radius_estimate] || 1.0;
    
    const geometry = new THREE.SphereGeometry(size, 32, 32);
    const material = new THREE.MeshPhongMaterial({
        color: color,
        emissive: color,
        emissiveIntensity: 0.2,
        shininess: 100
    });
    
    const mesh = new THREE.Mesh(geometry, material);
    mesh.position.set(position.x, position.y, position.z);
    
    // Add glow effect for critical nodes
    if (dpr.blast_radius_estimate === 'critical') {
        const glowGeometry = new THREE.SphereGeometry(size * 1.3, 32, 32);
        const glowMaterial = new THREE.MeshBasicMaterial({
            color: color,
            transparent: true,
            opacity: 0.2
        });
        const glow = new THREE.Mesh(glowGeometry, glowMaterial);
        mesh.add(glow);
    }
    
    return {
        mesh: mesh,
        data: dpr,
        originalColor: color
    };
}

// Create edge between nodes
function createEdge(from, to) {
    const points = [from, to];
    const geometry = new THREE.BufferGeometry().setFromPoints(points);
    const material = new THREE.LineBasicMaterial({
        color: 0x64b5f6,
        transparent: true,
        opacity: 0.3
    });
    
    return new THREE.Line(geometry, material);
}

// Create text label for node
function createLabel(text, position) {
    const canvas = document.createElement('canvas');
    const context = canvas.getContext('2d');
    canvas.width = 256;
    canvas.height = 64;
    
    context.fillStyle = 'rgba(100, 181, 246, 0.8)';
    context.font = 'Bold 24px Arial';

    const texture = new THREE.CanvasTexture(canvas);

// Handle node click
function onNodeClick(event) {
    const mouse = new THREE.Vector2();
    mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
    mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;
    
    const raycaster = new THREE.Raycaster();
    raycaster.setFromCamera(mouse, camera);
    
    const meshes = nodes.map(n => n.mesh);
    const intersects = raycaster.intersectObjects(meshes);
    
    if (intersects.length > 0) {
        const clickedMesh = intersects[0].object;
        const node = nodes.find(n => n.mesh === clickedMesh);
        
        if (node) {
            selectNode(node);
        }
    } else {
        deselectNode();
    }
}

## 🔍 Understanding DPRs

Each Decision Provenance Record captures:

- **Decision**: What was chosen
- **Rejected Alternatives**: What was considered but not chosen
- **Explicit Constraints**: Stated limitations
- **Implicit Assumptions**: Unstated assumptions the decision relies on
- **Intended Durability**: temporary | medium-term | foundational
- **Assumption Decay Risk**: low | medium | high
- **Blast Radius**: low | medium | high | critical
- **Active Workarounds**: Evidence the decision is causing pain
- **Causal Dependencies**: What this decision depends on
