 48 |     print(f"\n📥 Downloading {repo_url}...")

 53 |     # Try git clone first
 54 |     try:

 56 |             ["git", "clone", "--depth", str(depth), repo_url, str(target_dir)],

 60 |             print(f"   Cloned via git to {target_dir}")

 75 |     else:
