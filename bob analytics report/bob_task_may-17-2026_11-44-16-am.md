"""
Nexus Logging Configuration
Centralized logging setup with structured output.
"""
import logging
import sys
from pathlib import Path
from datetime import datetime
from typing import Optional

class ColoredFormatter(logging.Formatter):
    """Custom formatter with colors for console output."""
    
    COLORS = {
        'DEBUG': '\033[36m',    # Cyan
        'INFO': '\033[32m',     # Green
        'WARNING': '\033[33m',  # Yellow
        'ERROR': '\033[31m',    # Red
        'CRITICAL': '\033[35m', # Magenta
    }
    RESET = '\033[0m'
    
    def format(self, record):
        if sys.stdout.isatty():
            levelname = record.levelname
            if levelname in self.COLORS:
                record.levelname = f"{self.COLORS[levelname]}{levelname}{self.RESET}"
        return super().format(record)


def setup_logger(
    name: str = "nexus",
    level: str = "INFO",
    log_file: Optional[str] = None,
    console: bool = True
) -> logging.Logger:
    """
    Setup a logger with console and optional file output.
    
    Args:
        name: Logger name
        level: Log level (DEBUG, INFO, WARNING, ERROR, CRITICAL)
        log_file: Optional file path for log output
        console: Whether to output to console
    
    Returns:
        Configured logger instance
    """
    logger = logging.getLogger(name)
    logger.setLevel(getattr(logging, level.upper()))
    
    # Remove existing handlers
    logger.handlers.clear()
    
    # Console handler with colors
    if console:
        console_handler = logging.StreamHandler(sys.stdout)
        console_handler.setLevel(logging.DEBUG)
        console_formatter = ColoredFormatter(
            '%(asctime)s | %(levelname)-8s | %(name)s | %(message)s',
            datefmt='%H:%M:%S'
        )
        console_handler.setFormatter(console_formatter)
        logger.addHandler(console_handler)
    
    # File handler
    if log_file:
        log_path = Path(log_file)
        log_path.parent.mkdir(parents=True, exist_ok=True)
        
        file_handler = logging.FileHandler(log_path, encoding='utf-8')
        file_handler.setLevel(logging.DEBUG)
        file_formatter = logging.Formatter(
            '%(asctime)s | %(levelname)-8s | %(name)s | %(funcName)s:%(lineno)d | %(message)s',
            datefmt='%Y-%m-%d %H:%M:%S'
        )
        file_handler.setFormatter(file_formatter)
        logger.addHandler(file_handler)
    
    return logger

## 📖 Documentation

### Key Concepts

- **DPR (Decision Provenance Record)**: Captures a single architectural decision with its context, assumptions, and blast radius
- **CTG (Causal Temporal Graph)**: Directed graph showing how decisions depend on and influence each other
- **Assumption Decay**: When the implicit assumptions behind a decision become invalid over time
- **Blast Radius**: The scope of impact if a decision needs to change
- **Counterfactual**: Analysis of what would change if a different decision had been made

### Analysis Pipeline

1. **Layer 1**: Repository scanning and DPR generation (via LLM)
2. **Layer 2**: CTG edge generation (causal relationships)
3. **Layer 3**: Assumption decay monitoring
4. **Layer 4**: Counterfactual simulation
5. **Layer 5**: Knowledge concentration analysis
6. **Merge**: Aggregate all data into nexus_data.json

"""
Nexus Health Check Utilities
Comprehensive system health monitoring.
"""
import json
import sys
from pathlib import Path
from datetime import datetime, timezone
from typing import Dict, Any, List

def check_file_exists(path: Path, name: str) -> Dict[str, Any]:
    """Check if a file exists and return status."""
    exists = path.exists()
    size = path.stat().st_size if exists else 0
    return {
        "name": name,
        "path": str(path),
        "exists": exists,
        "size_kb": round(size / 1024, 2) if exists else 0,
        "status": "ok" if exists else "missing"
    }


def check_directory(path: Path, name: str) -> Dict[str, Any]:
    """Check if a directory exists and count files."""
    exists = path.exists()
    file_count = len(list(path.rglob("*"))) if exists and path.is_dir() else 0
    return {
        "name": name,
        "path": str(path),
        "exists": exists,
        "file_count": file_count,
        "status": "ok" if exists else "missing"
    }


def check_llm_config() -> Dict[str, Any]:
    """Check LLM provider configuration."""
    import os
    
    gemini_key = os.getenv("GEMINI_API_KEY") or os.getenv("GOOGLE_API_KEY")
    watsonx_key = os.getenv("WATSONX_API_KEY")
    watsonx_project = os.getenv("WATSONX_PROJECT_ID")
    
    providers = []
    issues = []
    
    if gemini_key:
        providers.append("gemini")
    if watsonx_key:
        if watsonx_project:
            providers.append("watsonx")
        else:
            issues.append("WATSONX_API_KEY set but WATSONX_PROJECT_ID missing")
    
    if not providers:
        issues.append("No LLM provider configured")
    
    return {
        "configured_providers": providers,
        "active_provider": providers[0] if providers else None,
        "gemini_configured": bool(gemini_key),
        "watsonx_configured": bool(watsonx_key and watsonx_project),
        "issues": issues,
        "status": "ok" if providers and not issues else "warning" if providers else "error"
    }


def check_neo4j() -> Dict[str, Any]:
    """Check Neo4j connection."""
    import os
    
    try:
        from neo4j import GraphDatabase
        
        uri = os.getenv("NEO4J_URI", "bolt://localhost:7687")
        user = os.getenv("NEO4J_USER", "neo4j")
        password = os.getenv("NEO4J_PASSWORD", "password")
        
        driver = GraphDatabase.driver(uri, auth=(user, password))
        driver.verify_connectivity()
        
        with driver.session() as session:
            result = session.run("MATCH (n) RETURN count(n) as node_count")
            node_count = result.single()["node_count"]
            
            result = session.run("MATCH ()-[r]->() RETURN count(r) as rel_count")
            rel_count = result.single()["rel_count"]
        
        driver.close()
        
        return {
            "connected": True,
            "uri": uri,
            "node_count": node_count,
            "relationship_count": rel_count,
            "status": "ok"
        }
    except ImportError:
        return {
            "connected": False,
            "error": "neo4j package not installed",
            "status": "warning"
        }
    except Exception as e:
        return {
            "connected": False,
            "error": str(e),
            "status": "error"
        }


def check_python_packages() -> Dict[str, Any]:
    """Check required Python packages."""
    required = [
        "fastapi",
        "uvicorn",
        "pydantic",
        "networkx",
        "reportlab",
    ]
    
    optional = [
        "neo4j",
        "google.genai",
        "ibm_watsonx_ai",
    ]
    
    installed = []
    missing = []
    
    for package in required:
        try:
            __import__(package.replace("-", "_"))
            installed.append(package)
        except ImportError:
            missing.append(package)
    
    optional_installed = []
    for package in optional:
        try:
            __import__(package.replace("-", "_").replace(".", "_"))
            optional_installed.append(package)
        except ImportError:
            pass
    
    return {
        "required_installed": installed,
        "required_missing": missing,
        "optional_installed": optional_installed,
        "status": "ok" if not missing else "error"
    }


def check_data_files() -> Dict[str, Any]:
    """Check critical data files."""
    root = Path(__file__).resolve().parent
    
    files = [
        check_file_exists(
            root / "postgres-analysis" / "output" / "nexus_layer1_dprs.json",
            "PostgreSQL Layer 1 DPRs"
        ),
        check_file_exists(
            root / "postgres-analysis" / "output" / "nexus_data.json",
            "PostgreSQL Merged Data"
        ),
        check_file_exists(
            root / "nexus-dashboard" / "index.html",
            "Dashboard HTML"
        ),
        check_file_exists(
            root / "nexus-dashboard" / "nexus_data.json",
            "Dashboard Data"
        ),
    ]
    
    ok_count = sum(1 for f in files if f["status"] == "ok")
    
    return {
        "files": files,
        "ok_count": ok_count,
        "total_count": len(files),
        "status": "ok" if ok_count == len(files) else "warning"
    }


def run_health_check() -> Dict[str, Any]:
    """Run comprehensive health check."""
    print("=" * 60)
    print("NEXUS HEALTH CHECK")
    print("=" * 60)
    
    checks = {
        "timestamp": datetime.now(timezone.utc).isoformat(),
        "python_version": f"{sys.version_info.major}.{sys.version_info.minor}.{sys.version_info.micro}",
        "platform": sys.platform,
    }
    
    # Check packages
    print("\n📦 Checking Python packages...")
    checks["packages"] = check_python_packages()
    if checks["packages"]["status"] == "ok":
        print(f"   ✅ All required packages installed ({len(checks['packages']['required_installed'])})")
    else:
        print(f"   ❌ Missing packages: {', '.join(checks['packages']['required_missing'])}")
    
    # Check LLM
    print("\n🤖 Checking LLM configuration...")
    checks["llm"] = check_llm_config()
    if checks["llm"]["status"] == "ok":
        print(f"   ✅ LLM configured: {checks['llm']['active_provider']}")
    else:
        print(f"   ⚠️  {', '.join(checks['llm']['issues'])}")
    
    # Check Neo4j
    print("\n🔗 Checking Neo4j...")
    checks["neo4j"] = check_neo4j()
    if checks["neo4j"]["connected"]:
        print(f"   ✅ Connected: {checks['neo4j']['node_count']} nodes, {checks['neo4j']['relationship_count']} relationships")
    else:
        print(f"   ⚠️  Not connected: {checks['neo4j'].get('error', 'Unknown error')}")
    
    # Check data files
    print("\n📁 Checking data files...")
    checks["data_files"] = check_data_files()
    print(f"   {checks['data_files']['ok_count']}/{checks['data_files']['total_count']} files found")
    for f in checks["data_files"]["files"]:
        icon = "✅" if f["status"] == "ok" else "❌"
        print(f"   {icon} {f['name']}: {f['size_kb']} KB" if f["exists"] else f"   {icon} {f['name']}: missing")
    
    # Overall status
    all_ok = all(
        checks[k].get("status") in ["ok", "warning"]
        for k in ["packages", "llm", "data_files"]
    )
    
    checks["overall_status"] = "healthy" if all_ok else "degraded"
    
    print(f"\n{'=' * 60}")
    if all_ok:
        print("✅ SYSTEM HEALTHY")
    else:
        print("⚠️  SYSTEM DEGRADED - Check issues above")
    print("=" * 60)
    
    return checks


def main():
    """Run health check and save report."""
    report = run_health_check()
    
    # Save report
    output_path = Path(__file__).resolve().parent / "health_check_report.json"
    with open(output_path, "w", encoding="utf-8") as f:
        json.dump(report, f, indent=2, default=str)
    
    print(f"\n💾 Report saved to: {output_path}")
    
    # Exit with error code if unhealthy
    sys.exit(0 if report["overall_status"] == "healthy" else 1)


if __name__ == "__main__":
    main()

class TestRunner:
    """Simple test runner for integration tests."""
    
    def __init__(self, base_url: str = "http://localhost:8000"):
        self.base_url = base_url
        self.passed = 0
        self.failed = 0
        self.tests = []
    
    def request(
        self,
        method: str,
        path: str,
        body: Optional[Dict[str, Any]] = None,
        expect_status: int = 200,
        timeout: int = 10
    ) -> Optional[Dict[str, Any]]:
        """Make HTTP request and return JSON response."""
        url = self.base_url + path
        
        try:
            if body:
                data = json.dumps(body).encode('utf-8')
                req = urllib.request.Request(
                    url,
                    data=data,
                    method=method,
                    headers={"Content-Type": "application/json"}
                )
            else:
                req = urllib.request.Request(url, method=method)
            
            response = urllib.request.urlopen(req, timeout=timeout)
            result = json.loads(response.read().decode('utf-8'))
            
            if response.status == expect_status:
                return result
            else:
                raise Exception(f"Expected {expect_status}, got {response.status}")
                
        except urllib.error.HTTPError as e:
            if e.code == expect_status:
                return None
            raise
    
    def test(self, name: str, func):
        """Run a test function."""
        try:
            start = time.time()
            func()
            duration = time.time() - start
            
            self.passed += 1
            self.tests.append({
                "name": name,
                "status": "PASS",
                "duration": duration
            })
            print(f"  ✅ {name} ({duration:.2f}s)")
            
        except Exception as e:
            self.failed += 1
            self.tests.append({
                "name": name,
                "status": "FAIL",
                "error": str(e)
            })
            print(f"  ❌ {name}: {e}")
    
    def summary(self):
        """Print test summary."""
        total = self.passed + self.failed
        print(f"\n{'=' * 60}")
        print(f"TEST SUMMARY: {self.passed}/{total} passed")
        if self.failed > 0:
            print(f"  ❌ {self.failed} failed")
        else:
            print(f"  ✅ All tests passed!")
        print("=" * 60)
        
        return self.failed == 0
